    79→    79→  create(): void {
    80→    80→    this.seed = this.readSeedParam();
    81→    81→    this.world = generateWorld(this.seed);
    82→    82→    const validation = validateWorld(this.world);
    83→    83→    if (!validation.ok) {
    84→    84→      console.warn('world validation issues:', validation.issues.slice(0, 4));
    85→    85→    }
    86→    86→    if (this.pendingSave) applySave(this.world, this.pendingSave);
    87→    87→
    88→    88→    this.buildWorldVisuals();
    89→    89→    this.drag = new DragController(this);
    90→    90→    this.skills = new SkillsController(this);
    91→    91→    this.drag.skills = this.skills;
    92→    92→    if (this.pendingSave) {
    93→    93→      this.skills.unlocked = { ...this.pendingSave.skills } as Record<SkillId, boolean>;
    94→    94→      for (const [k, v] of Object.entries(this.pendingSave.powers)) {
    95→    95→        const pw = this.skills.powers[k as PowerId];
    96→    96→        if (pw) {
    97→    97→          pw.unlocked = v.unlocked;
    98→    98→          pw.cooldownUntil = Math.max(0, v.cooldownUntil - Date.now());
    99→    99→          pw.activeUntil = 0;
   100→   100→        }
   101→   101→      }
   102→   102→      this.playSeconds = this.pendingSave.playSeconds ?? 0;
   103→   103→      const tray = this.pendingSave.tray;
   104→   104→      if (Array.isArray(tray)) {
   105→   105→        this.trayTapes = tray.slice(0, 5).map(v => (typeof v === 'number' ? v : null));
   106→   106→        while (this.trayTapes.length < 5) this.trayTapes.push(null);
   107→   107→        for (const id of this.trayTapes) if (id !== null) this.stowInTrayView(id);
   108→   108→      }
   109→   109→      this.pendingSave = null;
   110→   110→    } else if (this.carriedSkills) {
   111→   111→      this.skills.unlocked = { ...this.carriedSkills };
   112→   112→      this.skills.syncPowerUnlocks();
   113→   113→    }
   114→   114→
   115→   115→    this.setupCamera();
   116→   116→    this.setupInput();
   117→   117→    this.setupSaveHooks();
   118→   118→
   119→   119→    this.game.events.emit('tray:changed');
   120→   120→    this.game.events.emit('game:ready');
   121→   121→    this.emitCounters();
   122→   122→  }

   309→   309→    const cam = this.cameras.main;

   311→   311→    cam.scrollY += dy / cam.zoom;

   314→   314→  zoomBy(delta: number): void {
   315→   315→    const cam = this.cameras.main;
   316→   316→    this.zoomAt(this.scale.width / 2, this.scale.height / 2, cam.zoom + delta);
   317→   317→  }
   318→   318→
   319→   319→  /** zoom anchored on a screen point (cursor), then clamped by world bounds */
   320→   320→  zoomAt(screenX: number, screenY: number, targetZoom: number): void {
   321→   321→    const cam = this.cameras.main;
   322→   322→    const z = Phaser.Math.Clamp(targetZoom, ZOOM_MIN, ZOOM_MAX);
   323→   323→    if (z === cam.zoom) return;
   324→   324→    // pure math (getWorldPoint reads cached matrices around setZoom, which
   325→   325→    // silently cancels the anchor correction)
   326→   326→    const wx = cam.scrollX + screenX / cam.zoom;
   327→   327→    const wy = cam.scrollY + screenY / cam.zoom;
   328→   328→    cam.setZoom(z);
   329→   329→    cam.scrollX = wx - screenX / z;
   330→   330→    cam.scrollY = wy - screenY / z;
   331→   331→    this.game.events.emit('camera:zoom', z);
   332→   332→  }
   333→   333→
   334→   334→  fitToTable(): void {
   335→   335→    const cam = this.cameras.main;
   336→   336→    const z = Phaser.Math.Clamp(
   337→   337→      Math.min(this.scale.width / WORLD_W, this.scale.height / WORLD_H),
   338→   338→      ZOOM_MIN, ZOOM_MAX,
   339→   339→    );
   340→   340→    cam.setZoom(z);
   341→   341→    cam.centerOn(WORLD_W / 2, WORLD_H / 2);
   342→   342→    this.game.events.emit('camera:zoom', z);
   343→   343→  }

   345→   345→  // ------------------------------------------------------------------- input
   346→   346→  private setupInput(): void {
   347→   347→    this.input.mouse?.disableContextMenu();
   348→   348→    this.input.addPointer(3); // multi-touch
   349→   349→
   350→   350→    // wheel = zoom, centered on the cursor (v2; pan stays on keys/drag/touch)
   351→   351→    this.input.on('wheel', (_p: Phaser.Input.Pointer, _o: unknown, _dx: number, dy: number) => {
   352→   352→      audio.unlock();
   353→   353→      const d = Phaser.Math.Clamp(dy, -180, 180);
   354→   354→      this.zoomAt(_p.x, _p.y, this.cameras.main.zoom * Math.pow(0.9985, d));
   355→   355→    });
   356→   356→
   357→   357→    this.input.on('pointerdown', (pointer: Phaser.Input.Pointer) => {
   358→   358→      audio.unlock();
   359→   359→      if (this.uiBlocked || this.pointerOverUI(pointer)) return;
   360→   360→      this.drag.onPointerDown(pointer);
   361→   361→    });
   362→   362→    this.input.on('pointermove', (pointer: Phaser.Input.Pointer) => {
   363→   363→      this.drag.onPointerMove(pointer);
   364→   364→    });
   365→   365→    this.input.on('pointerup', (pointer: Phaser.Input.Pointer) => {
   366→   366→      this.drag.onPointerUp(pointer);
   367→   367→    });
   368→   368→
   369→   369→    const kb2 = this.input.keyboard as Phaser.Input.Keyboard.KeyboardPlugin;
   370→   370→    kb2.on('keydown-ESC', () => this.drag.cancel());
   371→   371→    kb2.on('keydown-D', () => this.skills.toggleScanner());
   372→   372→    kb2.on('keydown-M', () => this.skills.tryActivatePower('misplaced'));
   373→   373→    kb2.on('keydown-S', () => this.skills.tryActivatePower('matchScan'));
   374→   374→    kb2.on('keydown-G', () => this.skills.tryActivatePower('matchGather'));
   375→   375→    kb2.on('keydown-F', () => this.fitToTable());
   376→   376→    kb2.on('keydown', () => audio.unlock());
   377→   377→
   378→   378→    this.game.events.on('repair:closed', this.onRepairClosed, this);
   379→   379→    this.game.events.on('ui:block', (v: boolean) => { this.uiBlocked = v; });
   380→   380→    this.events.once(Phaser.Scenes.Events.SHUTDOWN, () => {
   381→   381→      this.game.events.off('repair:closed', this.onRepairClosed, this);
   382→   382→      this.game.events.off('ui:block');
   383→   383→    });
   384→   384→  }

   386→   386→  /** screen-space UI regions the game scene must not treat as table input */

   414→   414→  private readSeedParam(): number {
   415→   415→    const p = new URLSearchParams(window.location.search);
   416→   416→    if (this.override?.seed !== undefined) return this.override.seed >>> 0;
   417→   417→    if (this.override?.fresh) {
   418→   418→      if (this.override.carry && (this.override as { skills?: Record<string, boolean> }).skills) {
   419→   419→        this.carriedSkills = (this.override as { skills?: Record<SkillId, boolean> }).skills ?? null;
   420→   420→      }
   421→   421→      return randomSeed();
   422→   422→    }
   423→   423→    if (p.has('seed')) {
   424→   424→      const s = Number.parseInt(p.get('seed') as string, 10);
   425→   425→      if (Number.isFinite(s)) return s >>> 0;
   426→   426→    }
   427→   427→    // v1 save present? offer migration instead of silently dropping progress
   428→   428→    if (!p.has('autotest')) {
   429→   429→      try {
   430→   430→        const raw1 = localStorage.getItem('tidy-tapes-save-v1');
   431→   431→        if (raw1) {
   432→   432→          // flag only: the UI scene shows the choice panel when it becomes ready
   433→   433→          this.game.registry.set('v1SaveFound', true);
   434→   434→        }
   435→   435→      } catch { /* ignore */ }
   436→   436→    }
   437→   437→    // restore saved v2 run when present (autotest phase 2 uses ?restore=1)
   438→   438→    if ((!p.has('autotest') || p.has('restore')) && this.storageOk) {
   439→   439→      const save = loadRun();
   440→   440→      if (save) {
   441→   441→        this.pendingSave = save;
   442→   442→        return save.seed;
   443→   443→      }
   444→   444→    }
   445→   445→    return 20261118;
   446→   446→  }

   581→   581→  update(_time: number, delta: number): void {
   582→   582→    const dt = delta / 1000;
   583→   583→    this.playSeconds += dt;
   584→   584→
   585→   585→    // keyboard camera panning (works while holding tapes)
   586→   586→    let vx = 0, vy = 0;
   587→   587→    if (this.keys.A?.isDown || this.keys.LEFT?.isDown) vx -= 1;
   588→   588→    if (this.keys.D?.isDown || this.keys.RIGHT?.isDown) vx += 1;
   589→   589→    if (this.keys.W?.isDown || this.keys.UP?.isDown) vy -= 1;
   590→   590→    if (this.keys.S?.isDown || this.keys.DOWN?.isDown) vy += 1;
   591→   591→    if (vx !== 0 || vy !== 0) {
   592→   592→      this.panCamera(vx * PAN_SPEED * dt, vy * PAN_SPEED * dt);
   593→   593→    }
   594→   594→    if (this.keys.Q?.isDown) this.zoomBy(0.35 * dt);
   595→   595→    if (this.keys.E?.isDown) this.zoomBy(-0.35 * dt);
   596→   596→    if (this.keys.PLUS?.isDown) this.zoomBy(0.5 * dt);
   597→   597→    if (this.keys.MINUS?.isDown) this.zoomBy(-0.5 * dt);
   598→   598→
   599→   599→    this.drag.update();
   600→   600→    this.skills.update();
   601→   601→
   602→   602→    // autosave every 4s of play (debounced further on writes)
   603→   603→    this.saveTimer += dt;
   604→   604→    if (this.saveTimer > 4) {
   605→   605→      this.saveTimer = 0;
   606→   606→      if (this.drag.held === null && !this.drag.trayBatchDrag) this.save();
   607→   607→    }
   608→   608→  }

   612→   612→  debugShelve(count: number, combo?: number): number {
   613→   613→    let shelved = 0;
   614→   614→    for (const t of this.world.tapes) {
   615→   615→      if (shelved >= count) break;
   616→   616→      if (t.state !== 'loose' || (t.defect && !t.repaired)) continue;
   617→   617→      if (combo !== undefined && t.combo !== combo) continue;
   618→   618→      let row: typeof this.world.rows[number] | null = null;
   619→   619→      let slot = -1;
   620→   620→      for (const rowId of rowsOfGroup(t.combo)) {
   621→   621→        const r = this.world.rows[rowId];
   622→   622→        const k = r.slots.findIndex(x => x === null);
   623→   623→        if (k >= 0) { row = r; slot = k; break; }
   624→   624→      }
   625→   625→      if (!row || slot < 0) continue;
   626→   626→      this.seatTape(t.id, row.id, slot, true);
   627→   627→      shelved++;
   628→   628→    }
   629→   629→    this.sortedCount = this.world.tapes.filter(t2 => t2.state === 'shelved').length;
   630→   630→    this.skills.onShelved(this.sortedCount);
   631→   631→    this.emitCounters();
   632→   632→    this.checkCompletion();
   633→   633→    return shelved;
   634→   634→  }

   637→   637→  seatTape(id: number, rowId: number, slot: number, instant = false): void {
   638→   638→    const t = this.tapeData(id);
   639→   639→    if (!t) return;
   640→   640→    const row = this.world.rows[rowId];
   641→   641→    const view = this.tapeViews.get(id);
   642→   642→    t.state = 'shelved';
   643→   643→    t.row = rowId;
   644→   644→    t.slot = slot;
   645→   645→    t.rot = 0;
   646→   646→    row.slots[slot] = id;
   647→   647→    this.sortedCount = this.world.tapes.filter(tt => tt.state === 'shelved').length;
   648→   648→    if (view) {
   649→   649→      view.face.setVisible(false);
   650→   650→      if (view.overlay) view.overlay.setVisible(false);
   651→   651→      view.spine.setVisible(true);
   652→   652→      const p = slotPos(rowId, slot);
   653→   653→      const u = unitOfRow(rowId);
   654→   654→      const lr = rowId % ROWS_PER_UNIT;
   655→   655→      if (instant) {
   656→   656→        view.container.setPosition(p.x, p.y).setRotation(0).setDepth(unitPos(u).y + 1 + lr * 0.02);
   657→   657→      }
   658→   658→    }
   659→   659→    this.updateRowCount(rowId);
   660→   660→  }
